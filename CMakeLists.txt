cmake_minimum_required(VERSION 3.10)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
 if(DEFINED ENV{VITASDK})
  set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "VitaSDK toolchain")
 else()
  message(FATAL_ERROR "VITASDK is not set.")
 endif()
endif()
project(Lagi LANGUAGES C CXX)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
set(VITA_APP_NAME "Lagi")
set(VITA_TITLEID "LAGI00001")
set(VITA_VERSION "0.01")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(lagi src/main.cpp src/platform/vita/vita_platform.cpp src/platform/vita/vita_input.cpp src/platform/vita/vita_filesystem.cpp src/platform/vita/null_renderer.cpp src/platform/vita/null_audio.cpp)
target_include_directories(lagi PRIVATE include)
target_compile_definitions(lagi PRIVATE LAGI_PLATFORM_VITA=1 LAGI_NULL_RENDERER=1 LAGI_NULL_AUDIO=1)
target_link_libraries(lagi SceCtrl_stub SceDisplay_stub SceKernelThreadMgr_stub SceSysmodule_stub)
vita_create_self(lagi.self lagi UNSAFE)
vita_create_vpk(Lagi.vpk ${VITA_TITLEID} lagi.self VERSION ${VITA_VERSION} NAME ${VITA_APP_NAME})

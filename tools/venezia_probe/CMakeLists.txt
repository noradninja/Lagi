cmake_minimum_required(VERSION 3.19)
if(NOT DEFINED ENV{VITASDK})
  message(FATAL_ERROR "Set VITASDK in the environment")
endif()
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE FILEPATH "")
endif()
project(LagiVeneziaProbe C)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
set(VVNZRUNNER_SOURCE "" CACHE PATH "External VVNZRunner v2.1 checkout")
set(VVNZRUNNER_STUB "" CACHE FILEPATH "Exact v2.1 VitaSDK import archive")
set(MEP_ELF "" CACHE FILEPATH "Validated MeP probe ELF built separately")
foreach(file "${VVNZRUNNER_SOURCE}/VVNZRunner/include/vvnzrunner.h" "${VVNZRUNNER_STUB}" "${MEP_ELF}")
  if(NOT EXISTS "${file}" OR IS_DIRECTORY "${file}")
    message(FATAL_ERROR "Missing external dependency: ${file}")
  endif()
endforeach()
add_executable(venezia_probe arm_probe.c)
set_target_properties(venezia_probe PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
target_include_directories(venezia_probe PRIVATE "${VVNZRUNNER_SOURCE}/VVNZRunner/include")
target_link_libraries(venezia_probe "${VVNZRUNNER_STUB}" SceCtrl_stub SceIofilemgr_stub SceKernelThreadMgr_stub SceProcessmgr_stub SceSysmem_stub)
vita_create_self(venezia_probe.self venezia_probe UNSAFE)
vita_create_vpk(VeneziaProbe.vpk LAGIVNZ01 venezia_probe.self VERSION "01.00" NAME "Lagi Venezia Probe" FILE "${MEP_ELF}" mep_probe.elf)
